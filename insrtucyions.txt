*****************************************************************************

#!/bin/bash

# Update the Debian apt repositories, install OpenJDK 21

sudo apt update -y
sudo apt install fontconfig openjdk-21-jre -y
java -version

# A LTS (Long-Term Support) release is chosen 

sudo wget -O /etc/apt/keyrings/jenkins-keyring.asc \
  https://pkg.jenkins.io/debian-stable/jenkins.io-2026.key
echo "deb [signed-by=/etc/apt/keyrings/jenkins-keyring.asc]" \
  https://pkg.jenkins.io/debian-stable binary/ | sudo tee \
  /etc/apt/sources.list.d/jenkins.list > /dev/null
sudo apt update -y
sudo apt install jenkins -y

# Enable the Jenkins service to start at boot 

sudo systemctl enable jenkins

# Start the Jenkins service

sudo systemctl start jenkins

# Check the status of the Jenkins service 

sudo systemctl status jenkins

docker.sh 

*************************************************************************************************

#!/bin/bash

# Exit immediately if any command fails
set -e

echo "======================================"
echo " Docker Installation Started"
echo "======================================"

# Update package list
sudo apt update

# Install required packages
sudo apt install -y ca-certificates curl

# Create directory for Docker GPG key
sudo install -m 0755 -d /etc/apt/keyrings

# Download Docker GPG key
sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg \
-o /etc/apt/keyrings/docker.asc

# Set proper permissions
sudo chmod a+r /etc/apt/keyrings/docker.asc

# Add Docker Repository
echo \
"deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.asc] \
https://download.docker.com/linux/ubuntu \
$(. /etc/os-release && echo "$VERSION_CODENAME") stable" | \
sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# Update package index
sudo apt update

# Install Docker Engine
sudo apt install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

# Enable Docker Service
sudo systemctl enable docker

# Start Docker Service
sudo systemctl start docker

# Verify Docker Service
sudo systemctl status docker --no-pager

# Verify Installation
docker --version
docker compose version

echo "======================================"
echo " Docker Installed Successfully"
echo "======================================"

eksctl.sh

**************************************************************************************************************************
# Check the EC2 instance architecture
uname -m

# Detect the system architecture for eksctl
ARCH=$(uname -m | sed 's/x86_64/amd64/' | sed 's/aarch64/arm64/')

# Define the platform name
PLATFORM="$(uname -s)_${ARCH}"

# Download the latest official eksctl release
curl -fLO "https://github.com/eksctl-io/eksctl/releases/latest/download/eksctl_${PLATFORM}.tar.gz"

# Download the official checksum list
curl -fsSL https://github.com/eksctl-io/eksctl/releases/latest/download/eksctl_checksums.txt \
  -o eksctl_checksums.txt

# Verify the downloaded archive checksum
grep "${PLATFORM}.tar.gz" eksctl_checksums.txt | sha256sum --check

# Extract eksctl into /tmp
tar -xzf "eksctl_${PLATFORM}.tar.gz" -C /tmp

# Install eksctl into the system PATH
sudo install -m 0755 /tmp/eksctl /usr/local/bin/eksctl

# Verify the installed eksctl version
eksctl version

# Clean up downloaded installation files
rm -f "eksctl_${PLATFORM}.tar.gz" eksctl_checksums.txt

kubectl.sh

**************************************************************************************************************************

# Check the EC2 instance architecture
uname -m

# Download the latest stable kubectl version
VERSION=$(curl -fsSL https://dl.k8s.io/release/stable.txt)
curl -fLO "https://dl.k8s.io/release/${VERSION}/bin/linux/amd64/kubectl"

# Download the official SHA-256 checksum
curl -fLO "https://dl.k8s.io/release/${VERSION}/bin/linux/amd64/kubectl.sha256"

# Verify the downloaded binary before installing
echo "$(cat kubectl.sha256)  kubectl" | sha256sum --check

# Install kubectl into the system PATH
sudo install -o root -g root -m 0755 kubectl /usr/local/bin/kubectl

# Verify the installed client version
kubectl version --client

# Associate IAM OIDC provider with the EKS cluster
-------------------------------------------------------------------------------------------------------------------

eksctl utils associate-iam-oidc-provider \
    --region ap-south-1 \
    --cluster aniket-eks \
    --approve

The above command is crucial when setting up an EKS cluster because it enables IAM roles for service accounts (IRSA)
Amazon EKS uses OpenID Connect (OIDC) to authenticate Kubernetes service accounts with IAM roles.
Associating the IAM OIDC provider allows Kubernetes workloads (Pods) running in the cluster to assume IAM roles securely.
Without this, Pods in EKS clusters would require node-level IAM roles, which grant permissions to all Pods on a node.
Without this, these services will not be able to access AWS resources securely.
